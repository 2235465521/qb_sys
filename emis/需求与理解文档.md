        *   列5~7：`企标号`（来源企标）、`企标名`、`公司名称`（起草企业）。

        1. **查重/建档企业**：读取 `公司名称`，若系统不存在，则自动初始化该企业记录，补充缺失的组织维度。

### 5. PDF 实体资产管理：一键扫盘匹配与共享磁盘物理对齐
*   **共享磁盘物理根路径**：`Y:\磁盘阵列\标准文件下载\企标下载\`
*   **相对路径格式**：`整合/文件名.pdf`
*   **自动对齐机制**：

### 1. 共享磁盘 ZIP 打包文件寻址适配

改造 `backend/standards/services.py` 中的 `create_zip_from_standards` 函数，使其兼容 Django 原生 `media` 存储与物理共享磁盘 `Y:\` 路径：

```python
def create_zip_from_standards(standard_ids: list) -> bytes:
    import os
    from django.conf import settings
    
    standards = Standard.objects.filter(id__in=standard_ids).exclude(pdf_file='')
    shared_root = getattr(settings, 'SHARED_DISK_ROOT', r"Y:\磁盘阵列\标准文件下载\企标下载")
    
    buffer = io.BytesIO()
    with zipfile.ZipFile(buffer, 'w', zipfile.ZIP_DEFLATED) as zf:
        for std in standards:
            file_val = str(std.pdf_file)
            
            # 判断是否是保存在共享磁盘的相对路径（以 '整合/' 开头）
            if file_val.startswith('整合/'):
                file_path = os.path.join(shared_root, file_val)
            else:
                # 兼容本地 Django 原生 Media 存储
                file_path = std.pdf_file.path
                
            if os.path.exists(file_path):
                # 重新清洗压缩包内文件名，防止层级混乱
                arcname = f'{std.standard_no.replace("/", "_")}.pdf'
                zf.write(file_path, arcname=arcname)
            else:
                # 记录日志，忽略缺失文件
                continue
                
    buffer.seek(0)
    return buffer.getvalue()
```

### 2. 扫盘匹配相对路径模糊匹配核心算法

```python
def normalize_string(s: str) -> str:
    """去除所有非字母数字字符并转为小写，以便跨格式对齐标准号"""
    import re
    if not s:
        return ""
    # 去除扩展名
    if s.lower().endswith('.pdf'):
        s = s[:-4]
    return re.sub(r'[^a-zA-Z0-9]', '', s).lower()

def scan_and_align_pdf_assets() -> dict:
    """
    自动扫盘 `Y:\\磁盘阵列\\标准文件下载\\企标下载\\整合\\` 目录
    为缺失 pdf 路径的企标自动关联相对路径。
    """
    import os
    from django.conf import settings
    from standards.models import Standard
    
    shared_root = getattr(settings, 'SHARED_DISK_ROOT', r"Y:\磁盘阵列\标准文件下载\企标下载")
    target_dir = os.path.join(shared_root, "整合")
    
    if not os.path.exists(target_dir):
        return {"success": False, "matched_count": 0, "error": f"物理磁盘路径不存在: {target_dir}"}
        
    # 1. 读取所有的 PDF 文件并构建“归一化”映射表
    disk_files = [f for f in os.listdir(target_dir) if f.lower().endswith('.pdf')]
    normalized_file_map = {normalize_string(f): f for f in disk_files}
    
    # 2. 查询出所有没有关联 PDF 的企标
    unlinked_standards = Standard.objects.filter(type='enterprise', pdf_file='')
    matched_count = 0
    
    for std in unlinked_standards:
        std_norm = normalize_string(std.standard_no)
        # 支持双向子串包含匹配
        matched_filename = None
        
        # 优先精确查找
        if std_norm in normalized_file_map:
            matched_filename = normalized_file_map[std_norm]
        else:
            # 模糊子串查找（例如标准号 Q/XMBL 001 包含在文件名 Q_XMBL_001_纸箱 中）
            for file_norm, filename in normalized_file_map.items():
                if std_norm in file_norm or file_norm in std_norm:
                    matched_filename = filename
                    break
                    
        if matched_filename:
            # 存储相对路径：'整合/文件名.pdf'
            std.pdf_file = f"整合/{matched_filename}"
            std.save(update_fields=['pdf_file'])
            matched_count += 1
            
    return {"success": True, "matched_count": matched_count}
```

### 3. 多列批次（10列）Excel 智能关联解析服务

```python
def parse_batch_normative_excel(file_obj) -> dict:
    """
    兼容 batch-normative-ref-28 格式，自动建档起草企业、企标及规范引用链路
    """
    import pandas as pd
    from django.db import transaction
    from django.db.models import F
    from companies.models import Company
    from standards.models import Standard, NormativeReference
    
    result = {'companies_created': 0, 'standards_created': 0, 'citations_added': 0, 'errors': []}
    
    try:
        df = pd.read_excel(file_obj)
    except Exception as e:
        result['errors'].append(f"文件读取失败: {str(e)}")
        return result
        
    # 校验是否包含起步的核心列
    required = ['企标号', '公司名称', '企标中引用的标准号']
    if not all(col in df.columns for col in required):
        result['errors'].append("Excel 表头缺少核心列：'企标号', '公司名称', '企标中引用的标准号'")
        return result

    for idx, row in df.iterrows():
        row_idx = idx + 2  # Excel 行号 1-indexed (加上表头)
        
        company_name = str(row['公司名称']).strip()
        std_no = str(row['企标号']).strip()
        std_name = str(row['企标名']).strip() if '企标名' in df.columns and pd.notna(row['企标名']) else ''
        
        # 匹配被引用标准号（优先完整标准号，其次中引标准号）
        cited_no = None
        if '发布时引用的完整标准号' in df.columns and pd.notna(row['发布时引用的完整标准号']):
            cited_no = str(row['发布时引用的完整标准号']).strip()
        elif pd.notna(row['企标中引用的标准号']):
            cited_no = str(row['企标中引用的标准号']).strip()
            
        if not company_name or not std_no or not cited_no:
            continue
            
        try:
            with transaction.atomic():
                # 1. 自动对齐/建档企业
                company, created_co = Company.objects.get_or_create(
                    name=company_name,
                    defaults={
                        'credit_code': f'TEMP_{uuid.uuid4().hex[:14].upper()}',  # 临时信用代码
                        'status': 'active'
                    }
                )
                if created_co:
                    result['companies_created'] += 1
                    
                # 2. 自动对齐/建档企标
                clean_id = std_no.replace(" ", "").replace("/", "_")
                standard, created_std = Standard.objects.get_or_create(
                    standard_no=std_no,
                    defaults={
                        'clean_id': clean_id,
                        'title': std_name or '批量入库标准',
                        'company': company,
                        'type': 'enterprise',
                        'is_parsed': True
                    }
                )
                if created_std:
                    result['standards_created'] += 1
                else:
                    if not standard.is_parsed:
                        standard.is_parsed = True
                        standard.save(update_fields=['is_parsed'])
                        
                # 3. 关联或建档被引用的国标（统计引用热度）
                cited_std = Standard.objects.filter(standard_no=cited_no, type='national').first()
                
                # 4. 创建规范性引用关系记录
                ref, created_ref = NormativeReference.objects.get_or_create(
                    source_standard=standard,
                    cited_standard_no=cited_no,
                    defaults={'cited_standard': cited_std}
                )
                if created_ref:
                    result['citations_added'] += 1
                    # 累加国标引用热度计数
                    if cited_std:
                        Standard.objects.filter(pk=cited_std.pk).update(citation_count=F('citation_count') + 1)
                        
        except Exception as e:
            result['errors'].append(f"第 {row_idx} 行解析失败: {str(e)}")
            
    # 5. 后置自动扫盘匹配一次 PDF 相对路径，确保在导入完 Excel 后文件库直接对齐
    try:
        scan_res = scan_and_align_pdf_assets()
        result['pdf_aligned_count'] = scan_res.get('matched_count', 0)
    except Exception as scan_err:
        result['errors'].append(f"关联扫盘对齐 PDF 失败: {str(scan_err)}")
        
    return result
```

### 4. 会员字段自定义导出（Excel 生成逻辑）

```python
def export_members_custom(selected_fields: list) -> bytes:
    import openpyxl
    from users.models import Member
    
    # 字段汉化表映射
    FIELD_MAP = {
        'name': '姓名',
        'phone': '联系电话',
        'position': '职务',
        'company': '归属单位/组织',
        'notes': '备注说明',
        'created_at': '入库时间'
    }
    
    wb = openpyxl.Workbook()
    ws = wb.active
    ws.title = "会员自定义数据表"
    
    # 1. 写入勾选的表头
    headers = [FIELD_MAP.get(f, f) for f in selected_fields]
    ws.append(headers)
    
    # 2. 查询并写入会员行记录
    members = Member.objects.all().order_by('-created_at')
    for m in members:
        row_data = []
        for field in selected_fields:
            if field == 'name':
                row_data.append(m.name)
            elif field == 'phone':
                row_data.append(m.phone)
            elif field == 'position':
                # 级联获取职务
                roles = m.roles.all()
                positions = [r.position for r in roles if r.position]
                row_data.append(", ".join(positions) if positions else '暂无职位')
            elif field == 'company':
                row_data.append(m.company)
            elif field == 'notes':
                row_data.append(m.notes)
            elif field == 'created_at':
                row_data.append(m.created_at.strftime('%Y-%m-%d %H:%M:%S'))
            else:
                row_data.append('')
        ws.append(row_data)
        
    buffer = io.BytesIO()
    wb.save(buffer)
    buffer.seek(0)
    return buffer.getvalue()
```
